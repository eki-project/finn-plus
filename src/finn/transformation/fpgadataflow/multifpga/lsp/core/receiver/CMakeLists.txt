# C simulation testbenches of the LSP receiver components (built via ../CMakeLists.txt).
# One testbench per component file, in tb/. A file with several components gets one test per
# component, selected by the first argument. Exit code 77 marks a test that is not implemented yet.
function(receiver_tb file)
    add_executable(${file}_tb tb/${file}_tb.cpp)
    # The testbenches include the component headers next to this file
    target_include_directories(${file}_tb PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
    foreach(test IN LISTS ARGN)
        add_test(NAME ${file}.${test} COMMAND ${file}_tb ${test})
        set_tests_properties(${file}.${test} PROPERTIES SKIP_RETURN_CODE 77)
    endforeach()
endfunction()

receiver_tb(utils ctrl_status mac_dispatcher cmac_ingress cmac_egress)
receiver_tb(manager manager)
# One process per scenario, since connection() keeps its state in statics
receiver_tb(connection accept multi_beat l2_offset bad_fcs gap duplicate malformed buffer_full blocked
    unknown_stream probe output reassembly credit)
receiver_tb(controller_tx controller_tx)

# Receiver top function, generated by generate_receiver_top.py from a JSON config of the connections
# (format: see the script). Compiled as a check only; without a config it is skipped.
#   cmake -S . -B build -DSAMPLE_CONFIG=1                          # sample_config.json
#   cmake -S . -B build -DLSP_RECEIVER_CONFIG=/path/to/config.json
# The generator needs jinja2, e.g. from the FINN environment: -DPython3_EXECUTABLE=$(which python)
option(SAMPLE_CONFIG "Generate the receiver top from sample_config.json" OFF)
set(LSP_RECEIVER_CONFIG "" CACHE FILEPATH "JSON config of the receiver connections")
if(SAMPLE_CONFIG)
    set(receiver_config "${CMAKE_CURRENT_SOURCE_DIR}/sample_config.json")
elseif(LSP_RECEIVER_CONFIG)
    set(receiver_config "${LSP_RECEIVER_CONFIG}")
else()
    message(STATUS "No receiver config given (-DSAMPLE_CONFIG=1 or -DLSP_RECEIVER_CONFIG=...): skipping receiver_top")
    return()
endif()

find_package(Python3 REQUIRED COMPONENTS Interpreter)
execute_process(COMMAND "${Python3_EXECUTABLE}" -c "import jinja2" RESULT_VARIABLE jinja2_missing OUTPUT_QUIET ERROR_QUIET)
if(jinja2_missing)
    message(FATAL_ERROR "${Python3_EXECUTABLE} has no jinja2; pass -DPython3_EXECUTABLE=$(which python) from the FINN environment")
endif()
# Generated into the source dir so it can be opened and edited next to the headers; gitignored
set(receiver_top "${CMAKE_CURRENT_SOURCE_DIR}/receiver_top.cpp")
add_custom_command(
    OUTPUT "${receiver_top}"
    COMMAND Python3::Interpreter "${CMAKE_CURRENT_SOURCE_DIR}/generate_receiver_top.py"
        "${receiver_config}" "${receiver_top}"
    DEPENDS "${receiver_config}" generate_receiver_top.py receiver_top.cpp.j2
    COMMENT "Generating receiver_top.cpp from ${receiver_config}"
    VERBATIM
)
add_library(receiver_top OBJECT "${receiver_top}")
target_include_directories(receiver_top PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
